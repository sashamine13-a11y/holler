  1195→const translations: Record<Locale, Record<string, string>> = { en, ru, es, de };
  1196→
  1197→const STORAGE_KEY = 'holler_locale';
  1198→
  1199→export function detectLocale(): Locale {
  1200→  if (typeof window === 'undefined') return 'en';
  1201→  try {
  1202→    const stored = localStorage.getItem(STORAGE_KEY) as Locale | null;
  1203→    if (stored && translations[stored]) return stored;
  1204→    const browser = navigator.language?.slice(0, 2) as Locale | undefined;
  1205→    if (browser && translations[browser]) return browser;
  1206→  } catch { /* SSR / private mode */ }
  1207→  return 'en';
  1208→}
  1209→
  1210→export interface I18nContextValue {
  1211→  locale: Locale;
  1212→  /** Translate a key, falling back to English and then to the key itself. */
  1213→  t: (key: string) => string;
  1214→  /** Switch language globally (persisted in localStorage). */
  1215→  setLocale: (l: Locale) => void;
  1216→  /** BCP-47 tag to use for toLocaleDateString etc. */
  1217→  dateLocale: string;
  1218→}
  1219→
  1220→const I18nContext = createContext<I18nContextValue | null>(null);
  1221→
  1222→/** App-level provider — mount once in layout.tsx so every component shares one locale. */
  1223→export function I18nProvider({ children }: { children: ReactNode }) {
  1224→  const [locale, setLocaleState] = useState<Locale>('en');
  1225→
  1226→  useEffect(() => {
  1227→    setLocaleState(detectLocale());
  1228→  }, []);
  1229→
  1230→  const setLocale = useCallback((l: Locale) => {
  1231→    setLocaleState(l);
  1232→    try { localStorage.setItem(STORAGE_KEY, l); } catch { /* ignore */ }
  1233→  }, []);
  1234→
  1235→  const t = useCallback((key: string): string => {
  1236→    return translations[locale]?.[key] || translations.en[key] || key;
  1237→  }, [locale]);
  1238→
  1239→  const dateLocale = locale === 'ru' ? 'ru-RU'
  1240→    : locale === 'de' ? 'de-DE'
  1241→    : locale === 'es' ? 'es-ES'
  1242→    : 'en-GB';
  1243→
  1244→  const value = useMemo<I18nContextValue>(
  1245→    () => ({ locale, t, setLocale, dateLocale }),
  1246→    [locale, t, setLocale, dateLocale],
  1247→  );
  1248→
  1249→  return <I18nContext.Provider value={value}>{children}</I18nContext.Provider>;
  1250→}
  1251→
  1252→export function useI18n(): I18nContextValue {
  1253→  const ctx = useContext(I18nContext);
  1254→  if (!ctx) throw new Error('useI18n must be used within <I18nProvider>');
  1255→  return ctx;
  1256→}
